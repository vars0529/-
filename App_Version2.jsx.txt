function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function fmt(n, digits = 1) {
  return new Intl.NumberFormat("en-US", {
    minimumFractionDigits: digits,
    maximumFractionDigits: digits,
  }).format(n);
}

function StatCard({ icon: Icon, label, value, sub }) {
  return (
    <div className="rounded-3xl border border-white/10 bg-white/5 p-4 shadow-lg backdrop-blur">
      <div className="flex items-center gap-3 text-white/70">
        <div className="rounded-2xl bg-white/10 p-2">

        </div>
        <div className="text-xs uppercase tracking-[0.25em]">{label}</div>

      {sub ? <div className="mt-1 text-sm text-white/55">{sub}</div> : null}

}

function Progress({ label, value, max = 100, tone = "from-cyan-400 to-blue-500" }) {
  const pct = clamp((value / max) * 100, 0, 100);
  return (
    <div>
      <div className="mb-2 flex items-center justify-between text-sm text-white/70">
        <span>{label}</span>
        <span>{fmt(value, 0)}</span>
      </div>
      <div className="h-2 w-full overflow-hidden rounded-full bg-white/10">
        <div className={`h-full rounded-full bg-gradient-to-r ${tone}`} style={{ width: `${pct}%` }} />
      </div>
    </div>
  );
}

// safe clone helper: prefer structuredClone, fallback to JSON deep clone
const safeClone = (obj) => {
  try {
    if (typeof structuredClone === "function") return structuredClone(obj);
  } catch (e) {
    // ignore
  }
  return JSON.parse(JSON.stringify(obj));
};

// pure function: apply effect to a country object and return a new object
function applyEffectToCountryObj(c, effect) {
  if (!c) return c;
  return {
    ...c,
    treasury: +(c.treasury + (effect.treasury || 0)).toFixed(1),
    gdp: +(c.gdp + (effect.gdp || 0)).toFixed(3),
    tech: clamp(c.tech + (effect.tech || 0), 0, 100),
    military: clamp(c.military + (effect.military || 0), 0, 100),
    softPower: clamp(c.softPower + (effect.softPower || 0), 0, 100),
    stability: clamp(c.stability + (effect.stability || 0), 0, 100),
    publicSupport: clamp(c.publicSupport + (effect.publicSupport || 0), 0, 100),
    growth: +(c.growth + (effect.growth || 0)).toFixed(1),
    inflation: +(c.inflation + (effect.inflation || 0)).toFixed(1),
    debt: +(c.debt + (effect.debt || 0)).toFixed(1),
  };
}
